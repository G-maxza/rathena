// quest อาวุท เสท Crown of Good
//============================================================
-	script	#Tradeitem5	4_GEFFEN_08,{
	mes .NPC$;
	mes "สวัสดี!";
	mes "ต้องการแลกเปลี่ยนผสมของ ?";
	next;
	switch ( select(
		""+.sb$+" Weapon",
		""+.sb$+" ยกเลิก"
	)) {
		case 1:
			close2;
			callshop "Bartershop5",1;
			end;
		case 2:
			mes .NPC$;
			mes "แล้วพบกันใหม่.";
			close;
	}
	end;
	
OnInit:
	waitingroom "ร้านเเลกquest Item Weapon Good",0;
	end;
}
