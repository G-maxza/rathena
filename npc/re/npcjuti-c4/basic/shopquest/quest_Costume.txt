-	script	#Tradeitem30	4_GEFFEN_02,{
	mes .NPC$;
	mes "สวัสดี!";
	mes "ต้องการแลกเปลี่ยนผสมของ ?";
	next;
	switch ( select(
		""+.sb$+" CostumeGarment_PRO",
		""+.sb$+" ยกเลิก"
	)) {
		case 1:
			close2;
			callshop "Bartershop30",1;
			end;
		case 2:
			mes .NPC$;
			mes "แล้วพบกันใหม่.";
			close;
	}
	end;
	
OnInit:
	waitingroom "ร้านเเลกquest CostumeGarment PRO",0;
	end;
}
