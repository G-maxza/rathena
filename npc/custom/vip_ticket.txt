function	script	1d_vip_ticket	{
	// VIP 1 day.
	if (!vip_status(VIP_STATUS_EXPIRE)) {
		atcommand "@vip +1d "+strcharinfo(0)+"";
		#VIP_SYSTEM_TIME = gettimetick(2) + 86400;
		sc_start SC_VIP_SYSTEM,86400 * 1000,1;
		delitem "VIP_Ticket_1_Day",1;
		end;
	}
	if (#VIP_SYSTEM_TIME > gettimetick(2)) { 
		message strcharinfo(0),"VIP ของคุณจะหมดอายุใน: "+ gettimestr("%d-%m-%Y %H:%M:%S",21,vip_status(VIP_STATUS_EXPIRE)) +"";
		dispbottom "จนกว่าจะหมดเวลาจะไม่สามารถใช้อันใหม่ได้!";
		end;
	}
}

//==============================================
// Check & Update VIP Real Time.
//==============================================
-	script	TAURUS_CHK_VIP	-1,{
OnPCLoginEvent:
	// Check & Update VIP.
	if (#VIP_SYSTEM_TIME > gettimetick(2)) {
		sc_end SC_VIP_SYSTEM;
		sc_start SC_VIP_SYSTEM,((#VIP_SYSTEM_TIME - gettimetick(2))*1000),1;
	}
	// Check & End VIP.
	if (#VIP_SYSTEM_TIME <= gettimetick(2)) || (!vip_status(VIP_STATUS_EXPIRE)) {
		sc_end SC_VIP_SYSTEM;
		#VIP_SYSTEM_TIME -= #VIP_SYSTEM_TIME;
	}
	end;
}
