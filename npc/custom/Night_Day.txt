-	script	Day_Night_System	-1,{
	
	OnInit:
		.@hr = gettime(DT_HOUR);
	OnClock0600:
 	OnClock1800:
		if (.@hr >= 6 && .@hr < 18) { day; }
		else { night; }
	end;
}
