oasis,161,175,4	script	Power_Tw_Pl_Controller#1	3797,{
	set .@gm_level, 99; // ระดับ GM ต้องใช้เพื่อเริ่ม / หยุดกิจกรรม

	mes "[Power Twisted Plains]";
	mes "ยินดีต้อนรับสู่ Power Twisted Plains!";
	next;
	switch(select("^FF0000•^000000 เข้าสู่Power Twisted Plains.:" + ((getgmlevel() >= .@gm_level) ? "^FF0000•^000000 GM Menu." : "") + ":^FF0000•^000000 ยกเลิก.")) {
	case 1:
		if (.start != 1) { // ระยะเวลาการลงทะเบียน
			mes "[Power Twisted Plains]";
			mes "Power Twisted Plains ถูกปิดแล้ว.";
			mes "มันจะเปิดในเวลาเหล่านี้:";
			mes "03:00 และ 16.00"; // แสดงเวลาของคุณที่นี่
			close;
		}
		if (BaseLevel < 260 || Zeny < 1000000 || countitem(7227) < 10) {
			mes "คุณไม่สามารถเข้าสู่ Power Twisted Plains ได้เพราะไม่ตรงตามเงื่อนไข";
			mes "ข้อกำหนด:";
			mes " - Base Level ต้อง 260 ขึ้นไป";
			mes " - ใช้ Zeny 1,000,000";
			mes " - ใช้ไอเท็ม ID 7227 จำนวน 5 ชิ้น";
		}
		Zeny -= 1000000;
		delitem 7227, 10;
		announce strcharinfo(0) + " ได้เข้าสู่ Power Twisted Plains.", bc_npc;
		warp "ra_pol01", 294, 105;
		close;
	case 2:
		mes "[Power Twisted Plains]";
		switch (.start) { // event states
			case 0: mes "Power Twisted Plains ยังไม่ได้เริ่ม."; break;
			case 1: mes "Power Twisted Plains กำลังรับผู้เข้าร่วม."; break;
		}
		next;
		switch(select("^FF0000•^000000 เริ่มกิจกรรม.:^FF0000•^000000 หยุดกิจกรรม.:^FF0000•^000000 ยกเลิก.")) {
		case 1:
			mes "[Power Twisted Plains]";
			if (.start)
				mes "Power Twisted Plains เริ่มแล้ว.";
			else {
				mes "ที่เริ่มต้น Power Twisted Plains...";
				donpcevent strnpcinfo(0)+"::OnStartEvent";
			}
			close;
		case 2:
			mes "[Power Twisted Plains]";
			if (!.start)
				mes "Power Twisted Plains ยังไม่ได้เริ่ม.";
			else {
				mes "การหยุด Power Twisted Plains...";
				donpcevent strnpcinfo(0)+"::OnStopEvent";
			}
			close;
		case 3:
			break;
		}
		break;
	case 3:
		break;
	}
	mes "[Power Twisted Plains]";
	mes "ลาก่อน.";
	close;
	
OnClock0300:
	donpcevent strnpcinfo(0) + "::OnStartEvent";
	end;

OnClock0500:
	donpcevent strnpcinfo(0) + "::OnStopEvent";
	end;

OnClock1600:
	donpcevent strnpcinfo(0) + "::OnStartEvent";
	end;

OnClock1800:
	donpcevent strnpcinfo(0) + "::OnStopEvent";
	end;
	
OnSpawnMonsters:
	.start = 2;
	.round = 1;
	set .mob, 1; // ตั้งค่าจำนวนมอนสเตอร์ที่ต้องการ
	// สร้างมอนสเตอร์ ID 21950 จำนวน 30 ตัว
	set .@monster_count1, 1;
	for (set .@i, 0; .@i < .@monster_count1; set .@i, .@i + 1) {
		monster "ra_pol01", 0 + rand(0), 0 + rand(0), "Boss Monster",21950,1,strnpcinfo(0)+"::OnMonsterKilled";
	}
	end;
	
OnMonsterKilled:
	.mob--;
	if ( .mob == 25 || .mob == 5 )
		announce "Boss Monster"+ .mob +"Bossที่เหลืออยู่ในแผนที่.", bc_map;
	else if ( .mob <= 0 ) {
		announce "Boss Monster ถูกำจัดหมดเเล้ว", bc_map;
		goto OnSpawnMonsters;
	}
	end;
	
OnStartEvent:
	.start = 1;
	enablenpc "Power_Tw_Pl_Controller#1";
	disablenpc "เควสPower Twisted Plains#2"; // ปิด NPC ที่ใช้ในการออกจากกิจกรรม
	getmapxy .@map$, .@x, .@y, BL_NPC;
	mapwarp "ra_pol01", .@map$, .@x, .@y;
	announce "Power Twisted Plains จะเริ่มใน 5 นาที.", bc_all;
	sleep 60000*5;
	announce "Power Twisted Plains ได้เริ่มขึ้นแล้ว!", bc_all;
	disablenpc "Power_Tw_Pl_Controller#1"; // ปิด NPC ที่ใช้ในการเข้าร่วม
	donpcevent strnpcinfo(0) + "::OnSpawnMonsters"; // เรียกใช้งานฟังก์ชันเกิดมอนสเตอร์
	end;
	
OnStopEvent:
	announce "MPower Twisted Plainsถูกปิดแล้ว.", bc_all;
	killmonsterall "ra_pol01"; // ลบมอนสเตอร์ทั้งหมดในแผนที่
	getmapxy .@map$, .@x, .@y, BL_NPC;
	mapwarp "ra_pol01", .@map$, .@x, .@y;
	disablenpc "Power_Tw_Pl_Controller#1"; // ปิด NPC ที่ใช้ในการเข้าร่วม
	enablenpc "เควสPower Twisted Plains#2";
	.start = .round = .mob = 0; // รีเซ็ตสถานะของกิจกรรมทั้งหมด
	end;
	
OnInit:
	waitingroom "Power Twisted Plains เปิด 03:00 และ 16:00 น.",0;
	end;
}
