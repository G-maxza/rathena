		announce "Boss Monster"+ .mob +"Bossที่เหลืออยู่ในแผนที่.", bc_map;

	end;

	.start = .round = .mob = 0; // รีเซ็ตสถานะของกิจกรรมทั้งหมด
	end;

	end;
