oasis,159,206,4	script	Varmundt_Depth_Controller#1	10371,{

	mes "[Varmundt Depth]";
	mes "ยินดีต้อนรับสู่ Varmundt Depth!";
	next;
	switch(select("^FF0000•^000000 เข้าสู่Varmundt Depth.:" + ((getgmlevel() >= .@gm_level) ? "^FF0000•^000000 GM Menu." : "") + ":^FF0000•^000000 ยกเลิก.")) {
	case 1:
		if (.start != 1) { // ระยะเวลาการลงทะเบียน
			mes "[Varmundt Depth]";
			mes "Varmundt Depth ถูกปิดแล้ว.";
			mes "มันจะเปิดในเวลาเหล่านี้:";
			mes "03:00 และ 16.00"; // แสดงเวลาของคุณที่นี่
			close;
		}
		if (BaseLevel < 260 || Zeny < 1000000 || countitem(7227) < 10) {
			mes "คุณไม่สามารถเข้าสู่ Varmundt Depth ได้เพราะไม่ตรงตามเงื่อนไข";
			mes "ข้อกำหนด:";

		}
		Zeny -= 1000000;
		delitem 7227, 10;
		announce strcharinfo(0) + " ได้เข้าสู่ Varmundt Depth.", bc_npc;
		warp "bl_depth1", 180, 180;
		close;
	case 2:
		mes "[Varmundt Depth]";
		switch (.start) { // event states
			case 0: mes "Varmundt Depth ยังไม่ได้เริ่ม."; break;
			case 1: mes "Varmundt Depth กำลังรับผู้เข้าร่วม."; break;
		}
		next;
		switch(select("^FF0000•^000000 เริ่มกิจกรรม.:^FF0000•^000000 หยุดกิจกรรม.:^FF0000•^000000 ยกเลิก.")) {
		case 1:
			mes "[Varmundt Depth]";
			if (.start)
				mes "Varmundt Depth เริ่มแล้ว.";
			else {
				mes "ที่เริ่มต้น Varmundt Depth...";
				donpcevent strnpcinfo(0)+"::OnStartEvent";
			}
			close;
		case 2:
			mes "[Varmundt Depth]";
			if (!.start)
				mes "Varmundt Depth ยังไม่ได้เริ่ม.";
			else {
				mes "การหยุด Varmundt Depth...";
				donpcevent strnpcinfo(0)+"::OnStopEvent";
			}
			close;
		case 3:
			break;
		}
		break;
	case 3:
		break;
	}
	mes "[Varmundt Depth]";
	mes "ลาก่อน.";
	close;
	
OnClock0300:
	donpcevent strnpcinfo(0) + "::OnStartEvent";
	end;

OnClock0500:
	donpcevent strnpcinfo(0) + "::OnStopEvent";
	end;

OnClock1600:
	donpcevent strnpcinfo(0) + "::OnStartEvent";
	end;

OnClock1800:
	donpcevent strnpcinfo(0) + "::OnStopEvent";
	end;
	
OnSpawnMonsters:
	.start = 2;
	.round = 1;
	set .mob, 1; // ตั้งค่าจำนวนมอนสเตอร์ที่ต้องการ
	// สร้างมอนสเตอร์ ID 22155 จำนวน 30 ตัว
	set .@monster_count1, 1;
	for (set .@i, 0; .@i < .@monster_count1; set .@i, .@i + 1) {
		monster "bl_depth1", 0 + rand(0), 0 + rand(0), "Boss Monster",22155,1,strnpcinfo(0)+"::OnMonsterKilled";
	}
	end;
	
OnMonsterKilled:
	.mob--;
	if ( .mob == 25 || .mob == 5 )
		announce "Boss Monster"+ .mob +"Bossที่เหลืออยู่ในแผนที่.", bc_map;
	else if ( .mob <= 0 ) {
		announce "Boss Monster ถูกำจัดหมดเเล้ว", bc_map;
		goto OnSpawnMonsters;
	}
	end;
	
OnStartEvent:
	.start = 1;
	enablenpc "Varmundt_Depth_Controller#1";
	disablenpc "เควสVarmundt Depth#2"; // ปิด NPC ที่ใช้ในการออกจากกิจกรรม
	getmapxy .@map$, .@x, .@y, BL_NPC;
	mapwarp "bl_depth1", .@map$, .@x, .@y;
	announce "Varmundt Depth จะเริ่มใน 5 นาที.", bc_all;
	sleep 60000*5;
	announce "Varmundt Depth ได้เริ่มขึ้นแล้ว!", bc_all;
	disablenpc "Varmundt_Depth_Controller#1"; // ปิด NPC ที่ใช้ในการเข้าร่วม
	donpcevent strnpcinfo(0) + "::OnSpawnMonsters"; // เรียกใช้งานฟังก์ชันเกิดมอนสเตอร์
	end;
	
OnStopEvent:
	announce "Varmundt Depthถูกปิดแล้ว.", bc_all;
	killmonsterall "bl_depth1"; // ลบมอนสเตอร์ทั้งหมดในแผนที่
	getmapxy .@map$, .@x, .@y, BL_NPC;
	mapwarp "bl_depth1", .@map$, .@x, .@y;
	disablenpc "Varmundt_Depth_Controller#1"; // ปิด NPC ที่ใช้ในการเข้าร่วม
	enablenpc "เควสVarmundt Depth#2";
	.start = .round = .mob = 0; // รีเซ็ตสถานะของกิจกรรมทั้งหมด
	end;
	
OnInit:
	waitingroom "Varmundt Depth เปิด 03:00 และ 16:00 น.",0;
	end;
}
