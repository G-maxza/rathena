			if (@bought_quantity[.@g] == 1)	mes "เงินที่ท่านเตรียมมามีไม่พอ ต้องใช้เงินจำนวน  "+@zenyuse+" Zeny ในการแลก "+getitemname(@bought_nameid[.@g])+"";
			else if (@bought_quantity[.@g] >= 2)	mes "เงินที่ท่านเตรียมมามีไม่พอ ต้องใช้เงินจำนวน  "+(@zenyuse*@bought_quantity[.@g])+" Zeny ในการแลก "+getitemname(@bought_nameid[.@g])+" จำนวน ^0055FF"+@bought_quantity[.@g]+"^000000 ea.";
		}
		else if ((@bought_quantity[.@g] <= @itemgetmany) && (Zeny >= (@zenyuse*@bought_quantity[.@g]))){
			set .@r,0;	set .@up,0;
			for( set .@r,0; .@r < getarraysize(@itemuse); set .@r,.@r+1 ) {
				if(.@up >= getarraysize(@itemuse)) break;
				delitem @itemuse[.@up],@itemusemany[.@up]*@bought_quantity[.@g];
				mes "-> "+getitemname(@itemuse[.@up])+" ได้ถูกนำไปใช้จำนวน ^0055FF"+(@itemusemany[.@up]*@bought_quantity[.@g])+"^000000 ea.";
				set .@up,.@up+1;
				continue;
			}
			if(@broken != 0) {	set .@failmake,100-@broken;	mes "^FF0000 -> [มีโอกาสเสียหาย]  "+.@failmake+" % ^000000";	}
			if(@zenyuse != 0) {	
				set Zeny,(Zeny-@zenyuse*@bought_quantity[.@g]);
				mes "-> จำนวนเงินที่ต้องการ ^0055FF"+(@zenyuse*@bought_quantity[.@g])+"^000000 zeny";
				mes "-> ต้องการใช้  Craft ItemS จำนวน ^0055FF"+@bought_quantity+"^000000 ชิ้น...";
				mes "กำลังทำ "+getitemname(@bought_nameid[.@g])+" จำนวน ^0055FF"+@bought_quantity+"^000000 ชิ้น...";
			}
			next;
			if(@broken != 0) {	
				set .@RTget,rand(1,100);
				if(@broken < .@RTget) {
					emotion 23;
					specialeffect2 155;
					mes "ก๊อง ก๊อง ก๊อง ... !!";
					mes "การทำItem ล้มเหลว เสียใจด้วย ... !!";
					set @bought_quantity[.@g],0;
				}
				if(@broken >= .@RTget) {
					emotion 21;
					specialeffect2 154;
					mes "ก๊อง ก๊อง ก๊อง ... !!";
					mes "การทำItem เสร็จสมบูรณ์ ... !!";
				}
			} 
			if(@broken == 0) {
				emotion 33;
				specialeffect2 488;
			}
			getitem @bought_nameid[.@g],@bought_quantity[.@g];
		}	
}
dispbottom "=============================================";
deletearray @bought_nameid[0],getarraysize(@bought_nameid);
deletearray @bought_quantity[0],getarraysize(@bought_quantity);
deletearray @itemuse[0],getarraysize(@itemuse);
deletearray @itemusemany[0],getarraysize(@itemusemany);
set @broken,0;	set @zenyuse, 0;
end;
