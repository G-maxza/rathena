	end;

	killmonsterall "vis_h02";
	killmonsterall "vis_h03";

	if(!playerattached()
			|| isSkipRespawn)
	end;
	
	monster strcharinfo(3),0,0,"--ja--",$mvpIds[rand(getarraysize($mvpIds))],1,strnpcinfo(0) + "::OnBossDead";
	setunitdata $@mobid[0],UMOB_DYNAMIC,.@rank;
	end;

	.@tier = getarg(0,1);
	.@mapName$ = "vis_h0" + .@tier;
	.@rank = 1 + (.@tier * 2);
	
	for(.@i = 0; .@i < .spawn_amount; .@i++){
		monster .@mapName$,0,0,"--ja--",$mvpIds[rand(getarraysize($mvpIds))],1,strnpcinfo(0) + "::OnBossDead";
		setunitdata $@mobid[0],UMOB_DYNAMIC,.@rank;
	}
	
	return;

	.spawn_amount = 50;
