// Get 100% based wording from 10000 amount
function	script	F_PercentBase10000	{
	.@v = getarg(0,0);
	if(.@v < 10)
	return "0.0" + .@v + "%";
	else if(.@v < 100)
	return "0." + .@v + "%";
	else{
		.@percent = (.@v / 100);
		.@decimal = (.@v % 100) / 10;
		return .@percent + "." + .@decimal + "%";
	}
	
	return;
}

// Get grade naming
function	script	GetGradeName	{
	.@grade = getarg(0,0);
	return (.@grade == 4) ? "A" : (.@grade == 3) ? "B" : (.@grade == 2) ? "C" : (.@grade == 1) ? "D" : "-";
}

// Is town map?
function	script	IsTownMap	{
	.@mapName$ = strcharinfo(3);
	
	.@isTownMap = (
	(.@mapName$ == "prontera")		// Main town
	|| (.@mapName$ == "tra_fild")	// Training Ground
	|| (.@mapName$ == "pvp_y_1-1")	// PvP
	)
	;
	
	return .@isTownMap;
}

// Get THE BOX rewards
function	script	GetTheBoxItem	{
	.@overrideItemId = getarg(0,0);
	.@itemType = getarg(1,0);
	
	.@itemId = 501;
	
	if(.@overrideItemId <= 0){
		// Random reward
		.@rand = (.@itemType > 0) ? -1: rand(10000);
		
		// 40%
		if((.@rand >= 6000)
				|| (.@itemType == 1))
		.@itemId = $weaponIds[rand(getarraysize($weaponIds))];
		// 45%
		else if((.@rand >= 1500)
				|| (.@itemType == 2))
		.@itemId = $equipmentIds[rand(getarraysize($equipmentIds))];
		// 10%
		else if((.@rand >= 500)
				|| (.@itemType == 3))
		.@itemId = $costumeIds[rand(getarraysize($costumeIds))];
		// 2.5%
		else if((.@rand >= 250)
				|| (.@itemType == 4))
		.@itemId = $card2Ids[rand(getarraysize($card2Ids))];
		// 2.5%
		else if((.@itemType == 0)
				|| (.@itemType == 5))
		.@itemId = $enchant2Ids[rand(getarraysize($enchant2Ids))];
	}
	else
	.@itemId = .@overrideItemId;
	
	.@type = getiteminfo(.@itemId,ITEMINFO_TYPE);
	
	if((.@type == 4)
			|| (.@type == 5)
			|| (.@type == 12)){
		randomoptgroup(1);
		
		getitem3 .@itemId,1,1,0,0,0,0,0,0,.@opt_id,.@opt_value,.@opt_param;
	}
	else
	getitem .@itemId,1;

	return;
}

// Get time remaining wording from seconds
function	script	GetTimerWording	{
	.@s = getarg(0,0);
	
	if(.@s >= 3600)
	{
		.@minutes = .@s % 60;
		return (.@s / 3600) + " ชั่วโมง" + ((.@minutes > 0) ? " " + .@minutes + " นาที" : "");
	}
	else if(.@s >= 60){
		.@seconds = .@s % 60;
		return (.@s / 60) + " นาที" + ((.@seconds > 0) ? " " + .@seconds + " วินาที" : "");
	}
	else
	return .@s + " วินาที";
	
	return;
}

	,7620		// Enriched_Oridecon
	,6226		// HD_Bradium
	,1000332	// Etherdeocon
	,1000334	// Enriched_Etherdeocon
	,1000368	// Etel_Bradium
	,1000336	// HD_Etherdeocon
	,1000369	// HD_Etel_Bradium
	,6635		// Blacksmith_Blessing
	;
	
	.@rewardId = .@rewards[rand(getarraysize(.@rewards))];
	
	// Blacksmith_Blessing will get 1~10ea
	if(.@rewardId == 6635)
	.@rewardAmount = rand(1,10);

	return;
}
